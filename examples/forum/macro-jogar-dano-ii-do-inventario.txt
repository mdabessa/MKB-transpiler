$${

if(dano);

log(&fParei);

unset(dano);

stop;

else;

log(&aAtirando..);

set(dano);

#slot=9;

slot(9);

gui(inventory);wait(20ms);

for(#i,1,26);

wait(1ms);

slotclick(%#slot%,l,true);

key(use);inc(#slot,1);

next;

gui;

log(&fParei!);

unset(dano);stop;

endif;

}$$
