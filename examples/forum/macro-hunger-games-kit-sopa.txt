$${

if(x);

log("&4[Macro] OFF);

unset(x);

stop();

else;

log("&d[Macro] ONN");

set(x);

&criador = Phodalao

ACHIEVEMENTGET(Parabéns By %&criador% ,276);

do;

if(%HEALTH%<14);

pick(282);

key(use);

wait(10ms);

key(drop);

wait(30ms);

slot(1);

endif;

loop

endif;

}$$
