$${
// source: https://forum.craftlandia.com.br/xf/threads/macro-pratico-pvp-autoclick-autoarco.365863/


IF(autousing);

ACHIEVEMENTGET(Autoclick OFF,351:1);

UNSET(autousing);

STOP();

ELSE;

ACHIEVEMENTGET(Autoclick ON,351:10);

PICK(258);

PICK(279);

PICK(267);

PICK(276);

SET(autousing);

DO;

KEY(attack);

WAIT(48ms);

LOOP;

ENDIF;

}$$

$${

IF(arco);

ACHIEVEMENTGET(AutoArco OFF,397:1);

PICK(258);

PICK(279);

PICK(267);

PICK(276);

UNSET(arco);

STOP();

ELSE;

ACHIEVEMENTGET(AutoArco ON,397);

SET(arco);

PICK(261);

IF(ITEM=261);

DO;

DO(23);

KEY(use);

LOOP;

WHILE(DURABILITY>10);

ENDIF;

NEXT;

ENDIF;

PICK(276);

ACHIEVEMENTGET(Arco não disponível,397:1);

UNSET(arco);

STOP;

}$$
