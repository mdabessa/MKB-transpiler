pick(%&pa%);

if(%DURABILITY%<180);

Look(%&frente%);

look(,90);

key(use);

wait(200ms);

endif;

IF(%DURABILITY%<90);

unset(segundo);

stop();

endif;

wait(2);

do(%&del%);

Look(%&frente%);

Look(,30);

PICK(%&bloco%);

KEY(use);

wait(2ms);

pick(%&pa%);

KEY(attack);

inc(#pontos, 2);

loop;

if(segundo);

wait(1);

gui(inventory);

wait(1);

getslotitem(35,#id,#stack);

if(%#stack%=0);

gui();

else;

do;

wait(1);
