$${
// source: https://forum.craftlandia.com.br/xf/threads/macro-pesca-e-repara-1-7-2-atualizado.328541/


IF(a);

LOG("&3&l[Macro] &fPescaria finalizado. ");

UNSET(a);

STOP();

ELSE;

LOG("&3&l[Macro] &fPescaria iniciado. ");

SET(a);

&olhar=%DIRECTION%;

ifmatches(%&olhar%,N);

&olhar=NORTH;endif;

ifmatches(%&olhar%,S);

&olhar=SOUTH;endif;

ifmatches(%&olhar%,E);

&olhar=EAST;endif;

ifmatches(%&olhar%,W);

&olhar=WEST;endif;

do;

if(%DURABILITY%<30);

// Olhar para baixo

look(%&olhar%);

look(,90);

wait(5ms);

key(use);

wait(5ms);

key(use);

else;

// Olhar reto

look(%&olhar%);

look(,15);

PICK(346);

do;

KEY(USE);

WAIT(20000ms);

KEY(USE);

WAIT(200ms);

LOOP;

ENDIF;

while(a);

endif;

}$$
