// ~~ Macro para quebrar e vender esmeraldas ~~
// devido ao bug do abrir bau no alt tab, fazer com que o player
// pare de quebrar blocos, essa macro foi feita utilizando o sistemas de loja.
// é nescessario um segundo player vip, crie uma loja privada, para armazenar as esmeraldas,
// assim, o player principal, não precisa abrir baus para pegar as esmeraldas.
// Caso nescessario, utilizar uma macro para transferir constantemente os coins
// das contas mineradoras para a conta que vender/compra esmeraldas.

// CONFIG
// Nome da Macro
&macro_name = "tool";
// Id do bloco a ser minerado
#block_id = 129;
// Id do item dropado
#drop_id = 388;
// Nome da home da mineradora com os baus
&home = "esmeralda";
// blocos minerados
@#total_blocks = 0;
// blocos minerados, em tempo real o de cima é estimado, a cada ciclo completo, é atualizado para a real quantidade
@#real_total_blocks = -448;

// [Transpiler] toggle_macro(&macro_name, stop_macro=true)
stop_macro_scope_toggle_macro = true;

indexof(@&macros[],#index,"%&macro_name%");
//log("%@#macro_count%")
if(#index > -1);
	// remover macro_name da lista macros
	unset(&temp_array[]);
	foreach(@&macros[],&macro);
		if(&macro_name != &macro);
			push(&temp_array[],&macro);
		endif;
	next;

	unset(@&macros[]);
	foreach(&temp_array[],&value);
		push(@&macros[],&value);
	next;

	// [Transpiler] print(&text)
	&text_scope_print = "Macro &3%&macro_name% &f- &4OFF!";

	log("&f[&5 MACRO &f] %&text_scope_print%");

	keyup("forward");
	if(stop_macro_scope_toggle_macro);
		stop();
	endif;
else;
	push(@&macros[],"%&macro_name%");
	// [Transpiler] print(&text)
	&text_scope_print = "Macro &3%&macro_name% &f- &2ON!";

	log("&f[&5 MACRO &f] %&text_scope_print%");

endif;


// Iniciar a macro
slot(1);
wait(1t);
// Fast load, nao abrir invetario se ja tiver uma picareta na 1 slot,
// assim evitando bug de não quebrar bloco
if(%ITEM%!=278);
	// [Transpiler] set_inventory()

	// [Transpiler] print(&text)
	&text_scope_print = "Verificando inventario inicial...";

	log("&f[&5 MACRO &f] %&text_scope_print%");

	IF(%GUI%!="GUIINVENTORY");
		GUI("INVENTORY");
	ENDIF;

	#pickaxe_slot = 0;
	#rep_slot = 0;
	FOR(#slot,9,44);
		GETSLOTITEM(%#slot%,#item_id);
		// GETITEMINFO(%#item_id%,&item_name);

		IF(%#item_id%==278);
			if(#pickaxe_slot != 0);
				GUI();
				// [Transpiler] raise(&text)
				&text_scope_raise = "Mais de uma picareta no inventario!";

				log("&f[&4 MACRO - ERRO &f] %&text_scope_raise%");
				stop();

			endif;

			#pickaxe_slot = %#slot%;

		ELSEIF(%#item_id%==57);
			if(#rep_slot != 0);
				GUI();
				// [Transpiler] raise(&text)
				&text_scope_raise = "Mais de um slot de blocos de reparacao no inventario!";

				log("&f[&4 MACRO - ERRO &f] %&text_scope_raise%");
				stop();

			endif;
			#rep_slot = %#slot%;
		ELSEIF(%#item_id%==#block_id);
			if((#slot < 36) || (#slot > 44));
				GUI();
				// [Transpiler] raise(&text)
				&text_scope_raise = "Bloco de mineracao fora da hotbar!";

				log("&f[&4 MACRO - ERRO &f] %&text_scope_raise%");
				stop();

			endif;
		ELSEIF((%#item_id%!=0) && (%#item_id%!=#drop_id));
			GUI();
			// [Transpiler] raise(&text)
			&text_scope_raise = "Item invalido no inventario!";

			log("&f[&4 MACRO - ERRO &f] %&text_scope_raise%");
			stop();

		endif;
	NEXT;

	// Validar inventario
	if(#pickaxe_slot == 0);
		GUI();
		// [Transpiler] raise(&text)
		&text_scope_raise = "Sem picareta no inventario!";

		log("&f[&4 MACRO - ERRO &f] %&text_scope_raise%");
		stop();

	endif;

	if(#rep_slot == 0);
		// [Transpiler] warning(&text)
		&text_scope_warning = "Sem bloco de reparacao no inventario!";

		log("&f[&6 MACRO - AVISO &f] %&text_scope_warning%");

	endif;

	// Organizar inventario
	if(#pickaxe_slot != 36);
		SLOTCLICK(%#pickaxe_slot%,L,FALSE);
		WAIT(3t);
		SLOTCLICK("36",L,FALSE);
		WAIT(3t);
	endif;

	if(#rep_slot != 35);
		SLOTCLICK(%#rep_slot%,L,FALSE);
		WAIT(3t);
		SLOTCLICK("35",L,FALSE);
		WAIT(3t);
	endif;
	GUI();


	// Ir pra a home
	// [Transpiler] go_home(&gohome_home)
	&gohome_home_scope_go_home = "%&home%";

	#gohome_x = %XPOS%;
	#gohome_y = %YPOS%;
	#gohome_z = %ZPOS%;

	// [Transpiler] command(&command_cmd)
	&command_cmd_scope_command = "/home %&gohome_home_scope_go_home%";

	// [Transpiler] print(&text)
	&text_scope_print = "Comando: &3%&command_cmd_scope_command%";

	log("&f[&5 MACRO &f] %&text_scope_print%");

	echo("%&command_cmd_scope_command%");
	wait(1t);


	// aguardar o teleporte, maximo 100 ticks (5 segundos)
	DO(100);
		IF((%XPOS% != #gohome_x) || (%YPOS% != #gohome_y) || (%ZPOS% != #gohome_z));
			BREAK;
		ENDIF;
		wait(1t);
	LOOP;
	// #todo: verificar se o teleporte foi bem sucedido
	#pitch = %PITCH%;
	// Yaw é invertido
	#yaw = %YAW% - 180;
	if(#yaw < 0);
		#yaw = 360 + #yaw;
	elseif(#yaw > 360);
		#yaw = #yaw - 360;
	endif;

	wait(1t);

	// aguardar carregar o mundo
	do();
		look("355","90");
		wait(100ms);
		if(%HITID% != 0);
			BREAK;
		ENDIF;
	loop();
	wait(10t);
	look("%#yaw%","%#pitch%");
	wait(1t);

endif;

// desativar menu loja
// [Transpiler] command(&command_cmd)
&command_cmd_scope_command = "/menuloja off";

// [Transpiler] print(&text)
&text_scope_print = "Comando: &3%&command_cmd_scope_command%";

log("&f[&5 MACRO &f] %&text_scope_print%");

echo("%&command_cmd_scope_command%");
wait(1t);


do();
	#block_slot = 2;
	do();
		slot(%#block_slot%);
		wait(1t);
		if(%ITEM% != %#block_id%);
			keydown("jump");
			wait(1t);
			keyup("jump");
			wait(20t);
			if(%ITEM% != %#block_id%);
				INC(#block_slot, 1);
				if(#block_slot > 9);
					// vender itens
					// [Transpiler] print(&text)
					&text_scope_print = "Sem bloco na hotbar!";

					log("&f[&5 MACRO &f] %&text_scope_print%");

					// [Transpiler] sell_items()

					unset(#baus[]);
					#baus[0] = 307;
					#baus[1] = 326;
					#baus[2] = 4;
					#baus[3] = 39;
					#baus[4] = 57;


					// vender
					do(5);
						FOREACH(#baus[],#bau);
							look("270", "%#bau%");
							wait(1t);

							// placa
							if(%HITID% == 68);
								key("attack");
								wait(12t);
							endif;
						next;
					loop;

					break;
				endif;
			endif;
		endif;

		// [Transpiler] place_and_break()

		look("179","4");
		wait(1t);
		do(5);
			key("use");
			wait(1t);
			if((%YAW% != 179) || (%PITCH% != 4));
				looks("179","4");
				wait(1t);
			endif;
		loop;

		slot(1);
		do();
			c = false;
			do(150);
				key("attack");

				if(%HITID%==49);
					@#total_blocks = @#total_blocks + 4;
					c = true;
					break;
				endif;

				if((%YAW% != 179) || (%PITCH% != 4));
					looks("179","4");
					wait(0.2t);
				endif;
			loop;

			if(c);
				// [Transpiler] update_label()

				&label_text = "&aEsmeraldas: &3%@#total_blocks%";
				setlabel("ore_label","%&label_text%");

				break;
			else;
				do(5);
					log("&4KeyAttack bugado!!");
					wait(1t);
					playsound(note.harp);
					wait(10t);
				loop;
			endif;
		loop;


		// Checar durabilidade / reparar
		IF(%DURABILITY% < 450);
			// [Transpiler] raise(&text)
			&text_scope_raise = "&fFerramenta &4quebrando, &fparando macro...";

			log("&f[&4 MACRO - ERRO &f] %&text_scope_raise%");
			stop();

		elseif(%DURABILITY% < 460);
			// [Transpiler] print(&text)
			&text_scope_print = "Reparandot ferramenta...";

			log("&f[&5 MACRO &f] %&text_scope_print%");

			look("268","86");
			wait(1t);
			key("use");
			wait(1t);
		ENDIF;
	loop;

	// [Transpiler] fill_hotbar()

	// [Transpiler] print(&text)
	&text_scope_print = "Comprando minerios...";

	log("&f[&5 MACRO &f] %&text_scope_print%");

	unset(#baus[]);
	//#baus[0] = 310;
	//#baus[1] = 332;
	//#baus[2] = 1;
	//#baus[3] = 33;
	#baus[0] = 53;

	slot(8);
	wait(3t);
	if(%ITEM%==#block_id);
		break;
	endif;

	// ciclar até conseguir comprar
	do();
		FOREACH(#baus[],#bau);
			look("1", "%#bau%");
			wait(1t);

			// placa
			if(%HITID% == 68);
				key("use");
				wait(20t);
				slot(8);
				wait(20t);

				if(%ITEM%==#block_id);
					break;
				endif;
			endif;
		next;
		if(%ITEM%==#block_id);
			break;
		endif;

		wait(600t);
	loop;

	@#real_total_blocks = @#real_total_blocks + 448;
	@#total_blocks = @#real_total_blocks;
	slot(1);
	wait(5t);

loop;
