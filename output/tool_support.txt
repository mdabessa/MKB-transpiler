					GETSLOTITEM(%#slot__scope_pick_up_items%,#item_id_);
					if(#item_id_ != 0);
						is_inventory_full_scope_pick_up_items = true;
						break;
					endif;
				next;
			endif;

			gui();

			if(is_inventory_full_scope_pick_up_items);
				break;
			endif;
		next;

		wait(10t);

		// [Transpiler] go_home(&gohome_home_scope_go_home)
		&gohome_home_scope_go_home = "%&home_min%%#ind%";

		#gohome_y_scope_go_home = %YPOS%;
		#gohome_z_scope_go_home = %ZPOS%;

		&command_cmd_scope_command = "/home %&gohome_home_scope_go_home%";

		// [Transpiler] print(&text_scope_print)
		&text_scope_print = "Comando: &3%&command_cmd_scope_command%";

		log("&f[&5 MACRO &f] %&text_scope_print%");

		echo("%&command_cmd_scope_command%");
		wait(1t);


		// aguardar o teleporte, maximo 100 ticks (5 segundos)
		DO(100);
			IF((%XPOS% != #gohome_x_scope_go_home) || (%YPOS% != #gohome_y_scope_go_home) || (%ZPOS% != #gohome_z_scope_go_home));
				BREAK;
			ENDIF;
			wait(1t);
		LOOP;
		// #todo: verificar se o teleporte foi bem sucedido
		#pitch_scope_go_home = %PITCH%;
		// Yaw é invertido
		#yaw_scope_go_home = %YAW% - 180;
		if(#yaw_scope_go_home < 0);
			#yaw_scope_go_home = 360 + #yaw_scope_go_home;
		elseif(#yaw_scope_go_home > 360);
			#yaw_scope_go_home = #yaw_scope_go_home - 360;
		endif;

		wait(1t);

		// aguardar carregar o mundo
		do();
			look("355","90");
			wait(100ms);
			if(%HITID% != 0);
				BREAK;
			ENDIF;
		loop();
		wait(10t);
		look("%#yaw_scope_go_home%","%#pitch_scope_go_home%");
		wait(1t);

		wait(20t);
		// [Transpiler] fill_ores()

		// [Transpiler] print(&text_scope_print)
		&text_scope_print = "Enchendo baus de minerios das mineradoras...";

		log("&f[&5 MACRO &f] %&text_scope_print%");

		unset(#chests_scope_fill_ores[]);
		//#chests_scope_fill_ores[0] = 312;
		//#chests_scope_fill_ores[1] = 335;
		//#chests_scope_fill_ores[2] = 5;
		//#chests_scope_fill_ores[3] = 33;
		#chests_scope_fill_ores[0] = 53;

		FOREACH(#chests_scope_fill_ores[],#chest);
			is_chest_full_scope_fill_ores = false;

			look("181","%#chest%");
			wait(1t);

			if((%HITID% != 54) && (%HITID% != 146));
				// [Transpiler] raise(&text_scope_raise)
				&text_scope_raise = "Sem baus / Trap incompleta";

				log("&f[&4 MACRO - ERRO &f] %&text_scope_raise%");
				stop();

			endif;

			key("use");
			wait(10t);

			FOR(#slot,54,89);
				GETSLOTITEM(%#slot%,#item_id_);

				if(#miid == #item_id_);
					SLOTCLICK(%#slot%,L,TRUE);
					WAIT(1t);
				endif;
			next;

			GETSLOTITEM("53",#last_chest_slot);
			GETSLOTITEM("0",#first_chest_slot);

			if((#last_chest_slot != 0) && (#first_chest_slot != 0));
				is_chest_full_scope_fill_ores = true;
				break;
			endif;

			gui();
			WAIT(1t);

			// bau não está cheio e não tem mais itens para guardar
			if(!is_chest_full_scope_fill_ores);
				break;
			else;
				// [Transpiler] print(&text_scope_print)
				&text_scope_print = "Bau cheio";

				log("&f[&5 MACRO &f] %&text_scope_print%");

			endif;
		next;

		wait(10t);
	next;
	wait(200t);
loop;
